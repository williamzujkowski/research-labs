#!/bin/sh
# verify revised snippets; scratch only. Adds one benign script with an embedded quoted string.
set -u
mkdir -p /tmp/home; export HOME=/tmp/home npm_config_cache=/tmp/npmcache CANARY_OUTSIDE_DIR=/tmp/out
cp -r /lab/template /work/app && cd /work/app
git init -q
node -e 'const f="package.json";const p=JSON.parse(require("fs").readFileSync(f));p.scripts.preinstall="node -e \"console.log(\\\"quoted payload\\\")\"";require("fs").writeFileSync(f,JSON.stringify(p,null,2))'
grep preinstall package.json
echo "== inventory (all seven install-time lifecycle scripts)"
npm ci --ignore-scripts --offline --no-audit --no-fund >/dev/null
npm query ':attr(scripts, [preinstall]), :attr(scripts, [install]), :attr(scripts, [postinstall]), :attr(scripts, [prepublish]), :attr(scripts, [preprepare]), :attr(scripts, [prepare]), :attr(scripts, [postprepare])' | node -e 'for (const p of JSON.parse(require("fs").readFileSync(0))) console.log(p.location || "(root)", JSON.stringify(p.scripts))'
rm -rf node_modules .canary
echo "== trace"
strace -f -qq -s 4096 -e trace=execve -o /tmp/install.trace npm ci --offline --no-audit --no-fund >/dev/null
echo "-- old grep"
grep '"-c", .* = 0$' /tmp/install.trace | grep -o '"-c", "[^"]*"'
echo "-- new grep"
grep '"-c", .* = 0$' /tmp/install.trace | grep -o -E '"-c", "([^"\\]|\\.)*"'
